        function switchScreen(screenId, btnElement) {

            document.querySelectorAll('.screen-content').forEach(el => el.classList.add('hidden-screen'));

            document.getElementById(screenId).classList.remove('hidden-screen');

            

            document.querySelectorAll('.nav-item').forEach(el => {

                el.classList.remove('active-tab');

            });

            btnElement.classList.add('active-tab');

        }


        function setRole(role) {

            const btnStudent = document.getElementById('btn-student-mode');

            const btnParent = document.getElementById('btn-parent-mode');

            const displayName = document.getElementById('user-display-name');


            if (role === 'student') {

                btnStudent.className = "px-3 py-1 rounded-md text-xs font-semibold bg-blue-600 text-white transition";

                btnParent.className = "px-3 py-1 rounded-md text-xs font-semibold text-slate-400 hover:text-white transition";

                displayName.textContent = "Alex Johnson (Student)";

            } else {

                btnParent.className = "px-3 py-1 rounded-md text-xs font-semibold bg-rose-600 text-white transition";

                btnStudent.className = "px-3 py-1 rounded-md text-xs font-semibold text-slate-400 hover:text-white transition";

                displayName.textContent = "Parent View (Alex's Guard)";

                switchScreen('screen-7', document.querySelectorAll('.nav-item')[6]);

            }

        }


        function toggleTask(checkbox) {

            const item = checkbox.nextElementSibling;

            if (checkbox.checked) {

                item.classList.add('line-through', 'text-slate-400');

            } else {

                item.classList.remove('line-through', 'text-slate-400');

            }

        }


        function addNewTaskPrompt() {

            const taskText = prompt("Enter new task for Freshman Semester 1:");

            if (taskText) {

                const list = document.getElementById('home-task-list');

                const div = document.createElement('div');

                div.className = "flex items-center justify-between p-3 bg-slate-50 rounded-lg border border-slate-200";

                div.innerHTML = `

                    <div class="flex items-center space-x-3">

                        <input type="checkbox" onchange="toggleTask(this)" class="w-4 h-4 text-blue-600 rounded">

                        <span class="text-sm font-medium text-slate-700">${taskText}</span>

                    </div>

                    <span class="text-xs bg-slate-200 text-slate-700 px-2 py-0.5 rounded font-medium">New</span>

                `;

                list.prepend(div);

            }

        }
